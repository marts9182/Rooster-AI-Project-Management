```python
def kdp_assets_dir() -> Path:
    """Resolve the KDP project root for cross-project asset reads.

    Defaults to ../kdp-puzzle-press relative to this project root. Override
    via the KDP_ASSETS_DIR env var. Raises FileNotFoundError if the path
    does not exist on disk.
    """
    raw = os.environ.get("KDP_ASSETS_DIR")
    if raw:
        p = Path(raw).resolve()
    else:
        # config.py is at src/etsy_rooster/config.py — project root is parents[2].
        project_root = Path(__file__).resolve().parents[2]
        p = (project_root.parent / "kdp-puzzle-press").resolve()
    if not p.is_dir():
        raise FileNotFoundError(f"KDP_ASSETS_DIR resolves to {p!s}, which does not exist")
    return p
```

```python
"""ColoringNiche — the data Plan 2a's pipeline needs about one coloring book."""

from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path


def _default_hero_indices(design_count: int) -> tuple[int, ...]:
    """Three quartile-ish indices, 1-indexed and within [1..design_count]."""
    return (
        max(1, design_count // 4),
        max(1, design_count // 2),
        max(1, (3 * design_count) // 4),
    )


@dataclass(frozen=True)
class ColoringNiche:
    """Everything the Etsy coloring pipeline needs about one KDP book.

    All fields are required except hero_indices (which defaults to three
    quartile-spaced indices). Validation runs in __post_init__ — a niche
    object is either fully valid or construction fails.
    """

    book_id: str
    title: str
    subtitle: str
    intro: str
    theme_tags: list[str]
    design_count: int
    source_dir: Path
    hero_indices: tuple[int, ...] = field(default=())

    def __post_init__(self) -> None:
        if not self.book_id:
            raise ValueError("book_id must be non-empty")
        if not self.title:
            raise ValueError("title must be non-empty")
        if self.design_count <= 0:
            raise ValueError(f"design_count must be > 0, got {self.design_count}")
        if not self.theme_tags:
            raise ValueError("theme_tags must contain at least one tag")
        if not self.source_dir.is_dir():
            raise ValueError(f"source_dir does not exist: {self.source_dir}")
        # Default + validate hero_indices.
        if not self.hero_indices:
            # frozen dataclass: object.__setattr__ is the documented escape hatch
            object.__setattr__(self, "hero_indices", _default_hero_indices(self.design_count))
        for i in self.hero_indices:
            if i < 1 or i > self.design_count:
                raise ValueError(
                    f"hero_indices must be in [1..{self.design_count}], got {i}"
                )
```

```python
"""Read a KDP book module + its rendered PNGs and return a ColoringNiche.

This is the cross-project bridge. The Etsy project does NOT import any KDP
Python code — it parses the book module's text to extract constants (no
sys.path mucking, no side-effect imports of generators). Asset files are
read with PIL only to validate they are 1-bit B&W.
"""

from __future__ import annotations

import ast
from pathlib import Path
from typing import Any

from PIL import Image

from etsy_rooster.coloring.niche import ColoringNiche


class KdpAssetError(RuntimeError):
    """Raised when KDP-side assets or metadata don't match expectations."""


_REQUIRED_CONSTANTS = ("TITLE", "SUBTITLE", "INTRO", "DESIGN_COUNT", "THEME_TAGS")


def load_coloring_niche(*, kdp_root: Path, book_id: str) -> ColoringNiche:
    """Build a ColoringNiche from kdp-puzzle-press project state.

    Reads:
      - <kdp_root>/src/pocket_rooster_press/books/<module_name>.py
      - <kdp_root>/assets/processed/coloring/<book_id>/page_NN.png (count = DESIGN_COUNT)

    Raises:
      KdpAssetError when any required file is missing, count is wrong, or
      PNGs aren't 1-bit mode.
    """
    module_name = book_id.replace("-", "_")
    module_path = kdp_root / "src" / "pocket_rooster_press" / "books" / f"{module_name}.py"
    if not module_path.is_file():
        raise KdpAssetError(f"book module not found: {module_path}")

    constants = _extract_constants(module_path)
    for name in _REQUIRED_CONSTANTS:
        if name not in constants:
            raise KdpAssetError(
                f"book module {module_path.name} missing constant {name}"
            )

    asset_dir = kdp_root / "assets" / "processed" / "coloring" / book_id
    if not asset_dir.is_dir():
        raise KdpAssetError(f"asset directory not found: {asset_dir}")

    design_count = int(constants["DESIGN_COUNT"])
    for i in range(1, design_count + 1):
        page = asset_dir / f"page_{i:02d}.png"
        if not page.is_file():
            raise KdpAssetError(f"missing asset: {page.name} (expected {design_count} pages)")
        with Image.open(page) as img:
            if img.mode != "1":
                raise KdpAssetError(f"{page.name} is not 1-bit B&W (mode={img.mode!r})")

    return ColoringNiche(
        book_id=book_id,
        title=str(constants["TITLE"]),
        subtitle=str(constants["SUBTITLE"]),
        intro=str(constants["INTRO"]),
        theme_tags=list(constants["THEME_TAGS"]),
        design_count=design_count,
        source_dir=asset_dir,
    )


def _extract_constants(module_path: Path) -> dict[str, Any]:
    """Parse a Python file's top-level assignments without importing it.

    Only literal values (strings, ints, floats, lists, tuples) are returned;
    any non-literal RHS is silently skipped. This is intentional — we read
    metadata, not runtime values.
    """
    tree = ast.parse(module_path.read_text(encoding="utf-8"))
    out: dict[str, Any] = {}
    for node in tree.body:
        if isinstance(node, ast.Assign):
            for target in node.targets:
                if isinstance(target, ast.Name):
                    try:
                        out[target.id] = ast.literal_eval(node.value)
                    except (ValueError, SyntaxError):
                        continue
        elif isinstance(node, ast.AnnAssign) and isinstance(node.target, ast.Name):
            if node.value is not None:
                try:
                    out[node.target.id] = ast.literal_eval(node.value)
                except (ValueError, SyntaxError):
                    continue
    return out
```

```python
"""Compose a ColoringNiche + its PNGs into an Etsy-edition PDF.

Layout per page:
  Cover page (1): title centered, subtitle below, license note in footer
  Design pages (N): image centered with small margin, footer with brand + page N
US Letter (612 x 792 pt). All pages are 8.5x11 portrait. No blank pages.
"""

from __future__ import annotations

from pathlib import Path

from reportlab.lib.pagesizes import LETTER
from reportlab.lib.units import inch
from reportlab.pdfgen.canvas import Canvas

from etsy_rooster.coloring.niche import ColoringNiche

PAGE_W, PAGE_H = LETTER  # 612 x 792 pt
MARGIN = 0.5 * inch
FOOTER_Y = 0.35 * inch
FOOTER_FONT = "Helvetica"
FOOTER_FONT_SIZE = 8
COVER_TITLE_FONT = "Times-Bold"
COVER_TITLE_SIZE = 26
COVER_SUBTITLE_FONT = "Times-Italic"
COVER_SUBTITLE_SIZE = 14
COVER_LICENSE = (
    "For personal coloring use only. Not for resale or commercial use."
)


def build_etsy_pdf(niche: ColoringNiche, output_path: Path) -> None:
    """Write a 1+N page Etsy-edition PDF to output_path.

    Page 1: cover with title, subtitle, license footer.
    Pages 2..N+1: design image i with brand footer "Pocket Rooster Press · page i".
    """
    output_path.parent.mkdir(parents=True, exist_ok=True)
    c = Canvas(str(output_path), pagesize=LETTER)

    _draw_cover(c, niche)
    c.showPage()

    for i in range(1, niche.design_count + 1):
        _draw_design_page(c, niche, design_index=i)
        c.showPage()

    c.save()


def _draw_cover(c: Canvas, niche: ColoringNiche) -> None:
    # Title — wrap manually if long.
    c.setFont(COVER_TITLE_FONT, COVER_TITLE_SIZE)
    title_y = PAGE_H - 3.5 * inch
    c.drawCentredString(PAGE_W / 2, title_y, niche.title)

    # Subtitle.
    c.setFont(COVER_SUBTITLE_FONT, COVER_SUBTITLE_SIZE)
    c.drawCentredString(PAGE_W / 2, title_y - 0.6 * inch, niche.subtitle)

    # Footer: license + page indicator.
    c.setFont(FOOTER_FONT, FOOTER_FONT_SIZE)
    c.drawCentredString(PAGE_W / 2, FOOTER_Y, COVER_LICENSE)


def _draw_design_page(c: Canvas, niche: ColoringNiche, *, design_index: int) -> None:
    png = niche.source_dir / f"page_{design_index:02d}.png"
    # Available image area: page minus margins minus footer reservation.
    img_top = PAGE_H - MARGIN
    img_bottom = FOOTER_Y + 0.25 * inch
    img_left = MARGIN
    img_right = PAGE_W - MARGIN
    img_w = img_right - img_left
    img_h = img_top - img_bottom

    c.drawImage(
        str(png),
        img_left,
        img_bottom,
        width=img_w,
        height=img_h,
        preserveAspectRatio=True,
        anchor="c",
        mask="auto",
    )

    c.setFont(FOOTER_FONT, FOOTER_FONT_SIZE)
    c.drawCentredString(
        PAGE_W / 2,
        FOOTER_Y,
        f"Pocket Rooster Press · page {design_index}",
    )
```

```python
"""Compose ColoringNiche → 4 preview PNGs for the Etsy listing.

Outputs (in order):
  preview_grid_4x4.png   — 4x4 grid of 16 evenly-sampled designs
  preview_hero_01.png    — full-size render of niche.hero_indices[0]
  preview_hero_02.png    — full-size render of niche.hero_indices[1]
  preview_hero_03.png    — full-size render of niche.hero_indices[2]

Etsy's primary listing image is rank 1, which we want to be the grid (the
'what you get' shot). PublishOrchestrator only uploads previews[0], so
ensuring the grid sorts first by filename matters.
"""

from __future__ import annotations

from pathlib import Path

from PIL import Image

from etsy_rooster.coloring.niche import ColoringNiche

GRID_SIDE_PX = 1600       # final 4x4 grid size (square)
GRID_CELLS = 4            # 4x4 grid
GRID_GAP_PX = 12          # gap between cells
GRID_BG = (255, 255, 255)
HERO_LONG_SIDE_PX = 1600  # hero PNG longest side


def build_previews(niche: ColoringNiche, output_dir: Path) -> list[Path]:
    output_dir.mkdir(parents=True, exist_ok=True)
    paths: list[Path] = []
    paths.append(_build_grid(niche, output_dir))
    for i, idx in enumerate(niche.hero_indices, start=1):
        paths.append(_build_hero(niche, design_index=idx, n=i, output_dir=output_dir))
    return paths


def _build_grid(niche: ColoringNiche, output_dir: Path) -> Path:
    indices = _grid_indices(niche.design_count, GRID_CELLS * GRID_CELLS)
    cell_size = (GRID_SIDE_PX - (GRID_CELLS + 1) * GRID_GAP_PX) // GRID_CELLS
    grid = Image.new("RGB", (GRID_SIDE_PX, GRID_SIDE_PX), GRID_BG)
    for cell_idx, design_idx in enumerate(indices):
        row = cell_idx // GRID_CELLS
        col = cell_idx % GRID_CELLS
        x = GRID_GAP_PX + col * (cell_size + GRID_GAP_PX)
        y = GRID_GAP_PX + row * (cell_size + GRID_GAP_PX)
        src = niche.source_dir / f"page_{design_idx:02d}.png"
        with Image.open(src) as im:
            im = im.convert("RGB")
            im.thumbnail((cell_size, cell_size), Image.Resampling.LANCZOS)
            # Center within cell.
            ox = x + (cell_size - im.width) // 2
            oy = y + (cell_size - im.height) // 2
            grid.paste(im, (ox, oy))
    out = output_dir / "preview_grid_4x4.png"
    grid.save(out, format="PNG", optimize=True)
    return out


def _build_hero(
    niche: ColoringNiche, *, design_index: int, n: int, output_dir: Path
) -> Path:
    src = niche.source_dir / f"page_{design_index:02d}.png"
    with Image.open(src) as im:
        im = im.convert("RGB")
        im.thumbnail((HERO_LONG_SIDE_PX, HERO_LONG_SIDE_PX), Image.Resampling.LANCZOS)
        out = output_dir / f"preview_hero_{n:02d}.png"
        im.save(out, format="PNG", optimize=True)
        return out


def _grid_indices(design_count: int, cells: int) -> list[int]:
    """Evenly-spaced 1-indexed sample of `cells` indices from [1..design_count].

    If design_count < cells, repeats the last index to fill.
    """
    if design_count <= 0:
        return [1] * cells
    if design_count <= cells:
        return list(range(1, design_count + 1)) + [design_count] * (cells - design_count)
    step = design_count / cells
    return [max(1, min(design_count, int(round((i + 0.5) * step)))) for i in range(cells)]
```

```python
Run from the project root:
  python scripts/build_shop_assets.py

from __future__ import annotations

from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

# Palette B (locked in spec 2026-05-20).
TEAL = (31, 79, 102)      # #1F4F66
BRASS = (202, 164, 87)    # #CAA457
CREAM = (251, 243, 226)   # #FBF3E2
INK = (27, 27, 31)        # near-black

PROJECT_ROOT = Path(__file__).resolve().parent.parent
OUT_DIR = PROJECT_ROOT / "assets" / "shop"


def _load_serif_font(size: int) -> ImageFont.FreeTypeFont:
    """Best-effort load of a serif face installed on most systems."""
    candidates = [
        "C:/Windows/Fonts/georgia.ttf",
        "C:/Windows/Fonts/times.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSerif.ttf",
        "/System/Library/Fonts/Times.ttc",
    ]
    for path in candidates:
        try:
            return ImageFont.truetype(path, size=size)
        except OSError:
            continue
    return ImageFont.load_default()


def build_banner() -> Path:
    img = Image.new("RGB", (1200, 300), TEAL)
    draw = ImageDraw.Draw(img)

    font = _load_serif_font(56)
    text = "Pocket Rooster Press"
    bbox = draw.textbbox((0, 0), text, font=font)
    tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
    draw.text(((1200 - tw) // 2, (300 - th) // 2 - 6), text, fill=CREAM, font=font)

    # Tagline
    sub_font = _load_serif_font(20)
    sub = "Printables for makers · SVG cut files · Coloring · Printable wall art"
    bb2 = draw.textbbox((0, 0), sub, font=sub_font)
    sw = bb2[2] - bb2[0]
    draw.text(((1200 - sw) // 2, (300 + th) // 2 + 10), sub, fill=BRASS, font=sub_font)

    out = OUT_DIR / "banner.png"
    img.save(out, format="PNG", optimize=True)
    return out


def build_icon() -> Path:
    img = Image.new("RGB", (500, 500), CREAM)
    draw = ImageDraw.Draw(img)

    # Outer teal ring.
    draw.ellipse([16, 16, 484, 484], outline=TEAL, width=12)

    # PR monogram in brass serif.
    font = _load_serif_font(220)
    text = "PR"
    bbox = draw.textbbox((0, 0), text, font=font)
    tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
    draw.text(((500 - tw) // 2, (500 - th) // 2 - 30), text, fill=BRASS, font=font)

    out = OUT_DIR / "icon.png"
    img.save(out, format="PNG", optimize=True)
    return out


def main() -> None:
    OUT_DIR.mkdir(parents=True, exist_ok=True)
    banner = build_banner()
    icon = build_icon()
    print(f"wrote {banner}")
    print(f"wrote {icon}")


if __name__ == "__main__":
    main()
```

```python
    # 2. Build PDF + previews into tmp_path
    pdf_path = tmp_path / f"{BOOK_ID}.pdf"
    build_etsy_pdf(niche, pdf_path)
    preview_paths = build_previews(niche, tmp_path)
    assert pdf_path.is_file()
    assert len(preview_paths) == 4

    # 3. DB + SKU
    conn = sqlite3.connect(":memory:")
    db = CatalogDB(conn)
    db.init_schema()
    sku_id = db.create_sku(
        niche="coloring",
        params={
            "book_id": niche.book_id,
            "title": niche.title,
            "subtitle": niche.subtitle,
            "intro": niche.intro,
            "design_count": niche.design_count,
            "theme_tags": niche.theme_tags,
        },
    )
    db.attach_artifact_file(sku_id, kind="pdf", path=str(pdf_path))
    for p in preview_paths:
        db.attach_artifact_file(sku_id, kind="preview_png", path=str(p))

    # 4. Author metadata via real Gemini
    prompts_dir = (
        _PROJECT_ROOT / "src" / "etsy_rooster" / "listing_authoring" / "prompts"
    )
    summary = {
        "book_id": niche.book_id,
        "title": niche.title,
        "subtitle": niche.subtitle,
        "intro": niche.intro,
        "design_count": niche.design_count,
        "theme_tags": niche.theme_tags,
    }
    author = LLMListingAuthor(llm=GeminiListingClient(), prompts_dir=prompts_dir)
    draft = author.author(niche="coloring", artifact_summary=summary)
    db.set_listing_metadata(
        sku_id,
        title=draft.title,
        tags=draft.tags,
        description=draft.description,
        price_usd=draft.price_usd,
        materials=draft.materials,
    )

    # 5. Publish via real Etsy API
    tokens = TokenStore().load()
    etsy = EtsyClient(
        keystring=os.environ["ETSY_KEYSTRING"],
        shared_secret=os.environ["ETSY_SHARED_SECRET"],
        access_token=tokens["access_token"],
        shop_id=int(os.environ["ETSY_SHOP_ID"]),
    )
    orch = PublishOrchestrator(db=db, etsy=etsy, taxonomy_id=6343)
    listing_id = orch.publish(sku_id)

    # 6. Verify
    assert listing_id > 0
    assert db.current_state(sku_id) is SkuState.STAGED
    fetched = etsy.get_listing(listing_id)
    assert fetched["state"] == "draft"
    print(f"Created coloring draft listing {listing_id}: {draft.title!r}")
    print(f"View at: https://www.etsy.com/your/shops/PocketRoosterPress/tools/listings/state:draft")
```
